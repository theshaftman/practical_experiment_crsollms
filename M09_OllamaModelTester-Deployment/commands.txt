# Configure NginX sites available default routing
sudo nano /etc/nginx/sites-available/default

# Redirect HTTP to HTTPS
server {
    listen 80 default_server;
    listen [::]:80 default_server;
    server_name ollamamodeltester.dedyn.io;
    return 301 https://$server_name$request_uri;
}

# HTTPS server
server {
    listen 443 ssl default_server;
    listen [::]:443 ssl default_server;
    server_name ollamamodeltester.dedyn.io;

    # SSL Certificate paths (from Certbot)
    ssl_certificate /etc/letsencrypt/live/ollamamodeltester.dedyn.io/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/ollamamodeltester.dedyn.io/privkey.pem;

    # Proxy to Minikube frontend
    location / {
        proxy_pass http://MINIKUBE_IP:NodePort;   # Replace with your Minikube IP and NodePort
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

sudo nginx -t
sudo systemctl reload nginx
